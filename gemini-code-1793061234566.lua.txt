local Cache = {
    TouchEvents = {},

    RemoteFunctions = {},

    NPCs = {},

-- Utility
local function Log(msg)
    print("[SERVER BOMBARDMENT] " .. msg)
end

local function SafeCall(func)
    local success, err = pcall(func)
    if success then
        Cache.Stats.TotalFired = Cache.Stats.TotalFired + 1
    end
    return success
end

-- CRITICAL: Scan for ALL remotes that accept player/character arguments
local function ScanAllRemotes()
    Log("Scanning for FE remotes...")
    
    Cache.TouchEvents = {}
    Cache.AllRemoteEvents = {}
    Cache.RemoteFunctions = {}
    
    local locations = {
        Services.ReplicatedStorage,
        Services.ReplicatedFirst,
        Services.JointsService,
        Services.Workspace,
        Services.Players,
        Services.StarterGui
    }
    
    for _, location in ipairs(locations) do
        pcall(function()
            for _, obj in ipairs(location:GetDescendants()) do
                if obj:IsA("RemoteEvent") then
                    table.insert(Cache.AllRemoteEvents, obj)
                    
                    local nameLower = obj.Name:lower()
                    if nameLower:find("touch") or obj.Name == "TouchEvent" or nameLower:find("event") then
                        table.insert(Cache.TouchEvents, obj)
                    end
                elseif obj:IsA("RemoteFunction") then
                    table.insert(Cache.RemoteFunctions, obj)
                end
            end
        end)
    end
    
    Cache.Stats.RemotesFound = #Cache.AllRemoteEvents
    
    Log(string.format("Found %d RemoteEvents (%d TouchEvents) and %d RemoteFunctions", 
        #Cache.AllRemoteEvents, #Cache.TouchEvents, #Cache.RemoteFunctions))
end

-- Scan for trigger parts (parts that cause effects when touched)
local function ScanTriggerParts()
    Log("Scanning for trigger parts...")
    
    Cache.TriggerParts = {}
    
    for _, obj in ipairs(Services.Workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            local name = obj.Name:lower()
            
            -- Parts that typically trigger server events
            if name:find("trip") or name:find("trigger") or name:find("touch") or 
               name:find("seat") or name:find("slide") or name:find("platform") or
               name:find("button") or name:find("lever") or name:find("zone") or
               name:find("teleport") or name:find("damage") or name:find("kill") or
               obj:FindFirstChildOfClass("TouchTransmitter") then
                table.insert(Cache.TriggerParts, obj)
            end
        end
    end
    
    Log(string.format("Found %d trigger parts", #Cache.TriggerParts))
end

-- Get OTHER players (not yourself) - these are your targets
local function GetOtherPlayers()
    Cache.OtherPlayers = {}
    Cache.NPCs = {}
    
    -- Get all OTHER players (not you) if enabled
    if Config.TargetAllPlayers then
        for _, player in ipairs(Services.Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                local char = player.Character
                local hrp = char:FindFirstChild("HumanoidRootPart")
                local humanoid = char:FindFirstChildOfClass("Humanoid")
                
                if hrp and humanoid then
                    local playerData = {
                        Player = player,
                        Character = char,
                        HRP = hrp,
                        Humanoid = humanoid,
                        Head = char:FindFirstChild("Head"),
                        Torso = char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso"),
                        Backpack = player:FindFirstChild("Backpack"),
                        Parts = {}
                    }
                    
                    for _, part in ipairs(char:GetDescendants()) do
                        if part:IsA("BasePart") then
                            table.insert(playerData.Parts, part)
                        end
                    end
                    
                    table.insert(Cache.OtherPlayers, playerData)
                end
            end
        end
    end
    
    -- Get NPCs if enabled
    if Config.TargetNPCs then
        local npcFolders = {
            Services.Workspace:FindFirstChild("Customers"),
            Services.Workspace:FindFirstChild("NPCs"),
            Services.Workspace:FindFirstChild("Enemies"),
            Services.Workspace:FindFirstChild("Characters")
        }
        
        for _, folder in ipairs(npcFolders) do
            if folder then
                for _, npc in ipairs(folder:GetDescendants()) do
                    if npc:IsA("Model") then
                        local hrp = npc:FindFirstChild("HumanoidRootPart")
                        local humanoid = npc:FindFirstChildOfClass("Humanoid")
                        
                        if hrp and humanoid then
                            local npcData = {
                                Character = npc,
                                HRP = hrp,
                                Humanoid = humanoid,
                                Parts = {}
                            }
                            for _, part in ipairs(npc:GetDescendants()) do
                                if part:IsA("BasePart") then
                                    table.insert(npcData.Parts, part)
                                end
                            end
                            table.insert(Cache.NPCs, npcData)
                        end
                    end
                end
            end
        end
    end
    
    Cache.Stats.PlayersAffected = #Cache.OtherPlayers + #Cache.NPCs
    
    return Cache.Stats.PlayersAffected
end

-- THE MAIN ATTACK: Fire remotes with OTHER PLAYERS as the targets
local function FireRemotesOnOthers()
    if #Cache.OtherPlayers == 0 and #Cache.NPCs == 0 then
        return 0
    end
    
    local fireCount = 0
    local intensity = math.clamp(Config.ChaosIntensity or 3, 0, 5)
    
    -- Combine all targets based on configuration
    local allTargets = {}
    if Config.TargetAllPlayers then
        for _, p in ipairs(Cache.OtherPlayers) do table.insert(allTargets, p) end
    end
    if Config.TargetNPCs then
        for _, n in ipairs(Cache.NPCs) do table.insert(allTargets, n) end
    end
    
    local triggerParts = Cache.TriggerParts or {}
    
    -- METHOD 1: Fire TouchEvents with THEIR character and trigger parts
    for _, touchEvent in ipairs(Cache.TouchEvents) do
        for _, target in ipairs(allTargets) do
            for _, part in ipairs(triggerParts) do
                SafeCall(function()
                    touchEvent:FireServer(target.HRP, part)
                    fireCount = fireCount + 1
                end)
                SafeCall(function()
                    touchEvent:FireServer(target.Humanoid, part)
                    fireCount = fireCount + 1
                end)
            end
            SafeCall(function()
                touchEvent:FireServer(target.Character, "trip", true)
                fireCount = fireCount + 1
            end)
        end
    end
    
    -- METHOD 2: Fire ALL RemoteEvents with OTHER PLAYERS as arguments & scaled argument patterns
    for _, remote in ipairs(Cache.AllRemoteEvents) do
        for _, target in ipairs(allTargets) do
            SafeCall(function()
                remote:FireServer(target.HRP)
                fireCount = fireCount + 1
            end)
            
            SafeCall(function()
                remote:FireServer(target.Character)
                fireCount = fireCount + 1
            end)
            
            SafeCall(function()
                remote:FireServer(target.Humanoid)
                fireCount = fireCount + 1
            end)
            
            if target.Parts then
                for i = 1, math.min(2, #target.Parts) do
                    local bodyPart = target.Parts[i]
                    for j = 1, math.min(2, #triggerParts) do
                        local triggerPart = triggerParts[j]
                        SafeCall(function()
                            remote:FireServer(bodyPart, triggerPart)
                            fireCount = fireCount + 1
                        end)
                    end
                end
            end
            
            if target.Player then
                SafeCall(function()
                    remote:FireServer(target.Player)
                    fireCount = fireCount + 1
                end)
                
                SafeCall(function()
                    remote:FireServer(target.Player, true)
                    fireCount = fireCount + 1
                end)
                
                SafeCall(function()
                    remote:FireServer(target.Player.UserId)
                    fireCount = fireCount + 1
                end)
            end
            
            -- Scaled intensity patterns
            if intensity >= 1 then
                SafeCall(function()
                    remote:FireServer(target.HRP, "damage", 100, true)
                    fireCount = fireCount + 1
                end)
                SafeCall(function()
                    remote:FireServer(target.Character, "kill", target.Player or target.Character)
                    fireCount = fireCount + 1
                end)
            end
            
            if intensity >= 2 then
                SafeCall(function()
                    remote:FireServer(target.HRP, CFrame.new(0, 100, 0), "teleport")
                    fireCount = fireCount + 1
                end)
                SafeCall(function()
                    remote:FireServer(target.Character, "slide", "bounce", "reset")
                    fireCount = fireCount + 1
                end)
            end
            
            if intensity >= 3 then
                SafeCall(function()
                    remote:FireServer({
                        player = target.Player or target.Character,
                        character = target.Character,
                        humanoid = target.Humanoid,
                        serverSide = true,
                        FEOnly = true,
                        OnlyServer = true,
                        action = "wipe"
                    })
                    fireCount = fireCount + 1
                end)
            end
            
            if intensity >= 4 and target.Backpack then
                SafeCall(function()
                    remote:FireServer(target.Backpack, "activate", {serverSide = true})
                    fireCount = fireCount + 1
                end)
            end
            
            if intensity >= 5 then
                SafeCall(function()
                    remote:FireServer(target.Player, target.Character, target.Humanoid, "event", true, false, {serverSide = true, FEOnly = true, OnlyServer = true})
                    fireCount = fireCount + 1
                end)
            end
        end
    end
    
    -- METHOD 3: Invoke RemoteFunctions with OTHER PLAYERS
    for _, remoteFunc in ipairs(Cache.RemoteFunctions) do
        for _, target in ipairs(allTargets) do
            task.spawn(function()
                SafeCall(function()
                    remoteFunc:InvokeServer(target.HRP, "activate", {serverSide = true})
                    fireCount = fireCount + 1
                end)
            end)
            
            if intensity >= 3 and target.Player then
                task.spawn(function()
                    SafeCall(function()
                        remoteFunc:InvokeServer(target.Player, target.Character, "FEOnly", true)
                        fireCount = fireCount + 1
                    end)
                end)
            end
        end
    end
    
    return fireCount
end

function GUI.UpdateStats()
    if not GUI.RemoteStat then return end
    
    GUI.RemoteStat.Text = string.format("📡 Remotes: %d (%d Touch)", #Cache.AllRemoteEvents, #Cache.TouchEvents)
    GUI.TriggerStat.Text = string.format("🎯 Triggers: %d", #Cache.TriggerParts)
    GUI.TargetStat.Text = string.format("👥 Players: %d (+ %d NPCs)", #Cache.OtherPlayers, #Cache.NPCs)
    
    local formatted = Cache.Stats.TotalFired >= 1000000 and string.format("%.1fM", Cache.Stats.TotalFired / 1000000) or
                     Cache.Stats.TotalFired >= 1000 and string.format("%.1fK", Cache.Stats.TotalFired / 1000) or
                     tostring(Cache.Stats.TotalFired)
    
    GUI.FiredStat.Text = "💥 Fired: " .. formatted
end

function GUI.UpdateStatus()

    if Config.Active then
        GUI.Status.Text = "🔥 ACTIVE"
        GUI.Status.TextColor3 = Color3.fromRGB(255, 50, 50)
        GUI.Toggle.Text = "⏸️ STOP"
        GUI.Toggle.BackgroundColor3 = Color3.fromRGB(50, 150, 50)
    else
        GUI.Status.Text = "💤 INACTIVE"
        GUI.Status.TextColor3 = Color3.fromRGB(150, 150, 150)
        GUI.Toggle.Text = "🔥 START 🔥"
        GUI.Toggle.BackgroundColor3 = Color3.fromRGB(220, 20, 60)
    end
end

-- Main Loop with rate-limiting & bursts
task.spawn(function()
    while true do
        Services.RunService.Heartbeat:Wait()
        
        if Config.Active then
            GetOtherPlayers()
            
            -- Burst logic: fire remotes per second in bursts, respecting intensity
            local intensity = math.clamp(Config.ChaosIntensity or 3, 0, 5)
            local burstCount = math.random(15, 25) + (intensity * 5)
            
            for i = 1, math.min(burstCount, 30) do
                if not Config.Active then break end
                FireRemotesOnOthers()
                task.wait(0.02)
            end
            
            -- Cooldown between bursts: 0.4 - 0.6 seconds scaled by intensity
            local cooldown = math.random(40, 60) / 100 - (intensity * 0.03)
            cooldown = math.clamp(cooldown, 0.4, 0.6)
            task.wait(cooldown)
        end
    end
end)

-- Stats Loop

        end
    end
