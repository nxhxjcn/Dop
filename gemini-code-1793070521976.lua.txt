-- Get OTHER players with deep attribute, value object, animator, sound, and micro-component harvesting
local function GetOtherPlayers()

    Cache.NPCs = {}
    
    if Config.TargetAllPlayers then
        for _, player in ipairs(Services.Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                local char = player.Character
                local hrp = char:FindFirstChild("HumanoidRootPart")
                local humanoid = char:FindFirstChildOfClass("Humanoid")
                
                if hrp and humanoid then
                    local playerData = {
                        Player = player,
                        Character = char,
                        HRP = hrp,
                        Humanoid = humanoid,
                        Head = char:FindFirstChild("Head"),
                        Torso = char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso"),
                        Backpack = player:FindFirstChild("Backpack"),
                        Attributes = player:GetAttributes(),
                        CharAttributes = char:GetAttributes(),
                        Values = {},
                        Parts = {}
                    }
                    
                    -- Capture granular sub-components: BaseParts, Accessories, Attachments, Animators, Sounds, Highlights
                    for _, part in ipairs(char:GetDescendants()) do
                        if part:IsA("BasePart") or part:IsA("Accessory") or part:IsA("Attachment") or 
                           part:IsA("Clothing") or part:IsA("Animator") or part:IsA("Sound") or 
                           part:IsA("Highlight") or part:IsA("Constraint") then
                            table.insert(playerData.Parts, part)
                        elseif part:IsA("ValueBase") then
                            playerData.Values[part.Name] = part.Value
                        end
                    end
                    
                    if playerData.Backpack then
                        for _, tool in ipairs(playerData.Backpack:GetChildren()) do
                            table.insert(playerData.Parts, tool)
                        end
                    end
                    
                    -- Include player leaderstats or custom status folders if present
                    local statsFolder = player:FindFirstChild("leaderstats") or player:FindFirstChild("Status")
                    if statsFolder then
                        for _, val in ipairs(statsFolder:GetChildren()) do
                            if val:IsA("ValueBase") then
                                playerData.Values[val.Name] = val.Value
                            end
                        end
                    end
                    
                    table.insert(Cache.OtherPlayers, playerData)
                end
            end
        end
    end
    
    if Config.TargetNPCs then
        local npcFolders = {
            Services.Workspace:FindFirstChild("Customers"),
            Services.Workspace:FindFirstChild("NPCs"),
            Services.Workspace:FindFirstChild("Enemies"),
            Services.Workspace:FindFirstChild("Characters"),
            Services.Workspace:FindFirstChild("Mobs"),
            Services.Workspace:FindFirstChild("Dummies")
        }
        
        for _, folder in ipairs(npcFolders) do
            if folder then
                for _, npc in ipairs(folder:GetDescendants()) do
                    if npc:IsA("Model") then
                        local hrp = npc:FindFirstChild("HumanoidRootPart")
                        local humanoid = npc:FindFirstChildOfClass("Humanoid")
                        
                        if hrp and humanoid then
                            local npcData = {
                                Character = npc,
                                HRP = hrp,
                                Humanoid = humanoid,
                                Parts = {}
                            }
                            for _, part in ipairs(npc:GetDescendants()) do
                                if part:IsA("BasePart") or part:IsA("Accessory") or part:IsA("Animator") then
                                    table.insert(npcData.Parts, part)
                                end
                            end
                            table.insert(Cache.NPCs, npcData)
                        end
                    end
                end
            end
        end
    end
    
    Cache.Stats.PlayersAffected = #Cache.OtherPlayers + #Cache.NPCs
    return Cache.Stats.PlayersAffected
end
